const getSeneca = require('../../env/lambda/lambda')

function complete(seneca: any) {
  seneca.listen({type:'sqs',pin:'aim:alpha,run:job'})
  seneca.client({type:'sqs',pin:'aim:beta,push:thing'})

  const makeGatewayHandler = seneca.export('s3-store/makeGatewayHandler')
  seneca
    .act('sys:gateway,kind:lambda,add:hook,hook:handler', {
       handler: makeGatewayHandler('aim:alpha,run:job') })
}

exports.handler = async (
  event,
  context
) => {
  
  let seneca = await getSeneca('alpha', complete)
  
  let handler = seneca.export('gateway-lambda/handler')
  let res = await handler(event, context)
  return res
}
